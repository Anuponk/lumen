import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import {fileURLToPath,pathToFileURL} from 'node:url';
const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'..');
const read=file=>fs.readFileSync(path.join(root,'dist',file),'utf8');
const info=JSON.parse(read('validation-info.json'));
const build=JSON.parse(read('build-info.json'));
assert.equal(info.candidateSha,'65770fab037e611ad6d76f4503e15bc8dc1a5416');
assert.equal(build.sha,info.candidateSha);
assert.equal(info.environment,'isolated-validation');
assert.equal(info.backend,'browser-fixture');
assert.equal(info.productionData,false);
assert.equal(info.testProjectId,undefined,'No hosted database is used by this fixture');
const config=await import(pathToFileURL(path.join(root,'dist/src/persistence/config.js')));
assert.equal(config.LUMEN_SUPABASE_URL,'https://simulated-cloud.invalid');
assert.equal(config.LUMEN_SUPABASE_KEY,'simulated-client-no-network');
assert.ok(read('index.html').includes('<script src="/src/testing/validation-fixture.js"></script>'));
assert.doesNotMatch(read('index.html'),/@supabase|supabase-js/);
assert.ok(read('src/testing/validation-fixture.js').includes('Simulated cloud offline'));
for(const file of ['index.html','sw.js','manifest.webmanifest'])assert.ok(read(file).length>0);
console.log('Free simulated validation package verified; no live database or paid branch.');
